# Convert coordinate strings such as "70W" and "40N" into signed decimal degrees
convert_coord <- function(x) {
  
  value <- parse_number(x)
  direction <- str_extract(x, "[NSEW]")
  
  if_else(
    direction %in% c("S", "W"),
    -value,
    value
  )
}


# Clean the geographic lookup table and convert coordinate values
areas_clean <- areas_raw |>
  
  rename(
    ocean = Oc,
    area = Area,
    min_lat = MIN.lat,
    max_lat = MAX.lat,
    min_lon = MIN.lon,
    max_lon = MAX.lon
  ) |>
  
  mutate(
    min_lat = convert_coord(min_lat),
    max_lat = convert_coord(max_lat),
    min_lon = convert_coord(min_lon),
    max_lon = convert_coord(max_lon)
  )


# Calculate the total number of whales caught within each geographic area
area_summary <- whaling_clean |>
  
  group_by(ocean, area) |>
  
  summarise(
    total_catch = sum(total_catch, na.rm = TRUE),
    .groups = "drop"
  )


# Calculate catch totals for each operation type within each geographic area
area_operation <- whaling_clean |>
  
  group_by(ocean, area, operation_type) |>
  
  summarise(
    operation_catch = sum(total_catch, na.rm = TRUE),
    .groups = "drop"
  )


# Convert operation types into separate columns for each whaling area
area_operation_wide <- area_operation |>
  
  pivot_wider(
    names_from = operation_type,
    values_from = operation_catch,
    values_fill = 0
  ) |>
  
  # Rename columns so they are simple to use later in Google Earth Engine
  rename(
    aboriginal_catch = Aboriginal,
    commercial_catch = Commercial,
    illegal_catch = Illegal,
    special_permit_catch = `Special permit`,
    unknown_catch = Unknown
  )
  
# Identify the operation type responsible for the largest catch in each area
area_dominant <- area_operation |>
  
  group_by(ocean, area) |>
  
  slice_max(
    order_by = operation_catch,
    n = 1,
    with_ties = FALSE
  ) |>
  
  ungroup() |>
  
  rename(
    dominant_operation = operation_type
  ) |>
  
  select(
    ocean,
    area,
    dominant_operation
  )


# Combine total catch, operation breakdown, dominant operation and geographic boundaries
whaling_gee <- area_summary |>
  
  left_join(
    area_operation_wide,
    by = c("ocean", "area")
  ) |>
  
  left_join(
    area_dominant,
    by = c("ocean", "area")
  ) |>
  
  left_join(
    areas_clean,
    by = c("ocean", "area")
  ) |>
  
  # Calculate the percentage contribution of each operation type
  mutate(
    aboriginal_pct = round((aboriginal_catch / total_catch) * 100, 1),
    commercial_pct = round((commercial_catch / total_catch) * 100, 1),
    illegal_pct = round((illegal_catch / total_catch) * 100, 1),
    special_permit_pct = round((special_permit_catch / total_catch) * 100, 1),
    unknown_pct = round((unknown_catch / total_catch) * 100, 1)
  ) |>
  
  # Arrange columns into a logical order for Google Earth Engine
  select(
    area,
    ocean,
    total_catch,
    dominant_operation,
    
    aboriginal_catch,
    aboriginal_pct,
    
    commercial_catch,
    commercial_pct,
    
    illegal_catch,
    illegal_pct,
    
    special_permit_catch,
    special_permit_pct,
    
    unknown_catch,
    unknown_pct,
    
    min_lat,
    max_lat,
    min_lon,
    max_lon
  )

# Summarize cumulative catch for each species and operation type
species_operation_summary <- whaling_tidy |>
  
  group_by(species, operation_type) |>
  
  summarise(
    total_catch = sum(catch, na.rm = TRUE),
    .groups = "drop"
  ) |>
  
  # Calculate the total catch for each species so bars can be ordered by magnitude
  group_by(species) |>
  
  mutate(
    species_total = sum(total_catch)
  ) |>
  
  ungroup()
